# Volume Mount Watcher Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Live incremental sync of filesystem directories into the knowledge store via `ctxt watch`, with fsnotify (native) and polling fallback (Docker/NFS).

**Architecture:** New `internal/watcher/` package owns the Manager (one goroutine per watch). DB stores watch configs and per-file records (migration 008). Manager is a third goroutine in dpkms serve errgroup. CLI mirrors `feed.go` pattern.

**Tech Stack:** Go, fsnotify (promote from indirect dep), SHA-256 content hashing, SQLite (new tables via migration 008).

```go
package watcher

import (
    "path/filepath"
    "strings"
)

// MatchGlob reports whether path matches the given glob pattern.
// "**" is treated as matching any number of path segments (including zero).
func MatchGlob(pattern, path string) bool {
    // Normalise separators.
    path = filepath.ToSlash(path)
    pattern = filepath.ToSlash(pattern)

    // Fast-path: no double-star, delegate directly.
    if !strings.Contains(pattern, "**") {
        ok, _ := filepath.Match(pattern, path)
        return ok
    }

    // Split on "**" segments and match each chunk.
    return matchDoublestar(pattern, path)
}

// matchDoublestar handles patterns containing "**".
func matchDoublestar(pattern, path string) bool {
    // Expand "**" to match every possible sub-path prefix.
    parts := strings.Split(path, "/")
    return matchDS(pattern, parts)
}

func matchDS(pattern string, parts []string) bool {
    if pattern == "**" {
        return true
    }
    for i := range parts {
        candidate := strings.Join(parts[i:], "/")
        // Replace ** with the concrete prefix up to i.
        p := strings.Replace(pattern, "**", strings.Join(parts[:i], "/"), 1)
        p = strings.TrimPrefix(p, "/")
        if ok, _ := filepath.Match(p, candidate); ok {
            return true
        }
    }
    // Also try matching the whole path (zero segments consumed by **).
    p := strings.Replace(pattern, "**/", "", 1)
    if ok, _ := filepath.Match(p, strings.Join(parts, "/")); ok {
        return true
    }
    return false
}

// MatchAny reports whether path matches any of the given glob patterns.
// If patterns is empty, it returns true (match all).
func MatchAny(patterns []string, path string) bool {
    if len(patterns) == 0 {
        return true
    }
    for _, p := range patterns {
        if MatchGlob(p, path) {
            return true
        }
    }
    return false
}
```

package watcher

import (
    "context"
    "sync"

    "github.com/ideacrafterslabs/ctxt/internal/service"
    "github.com/ideacrafterslabs/ctxt/internal/storage"
)

// Ingester is the subset of service.Service used by the Manager.
type Ingester interface {
    Analyze(ctx context.Context, req service.AnalyzeRequest) (string, error)
    DeleteObject(ctx context.Context, id string) error
}

// Manager owns all active watch goroutines.
type Manager struct {
    store    storage.WatchStore
    ingester Ingester
    cancels  map[string]context.CancelFunc
    mu       sync.Mutex
}

// NewManager creates a Manager. store may be nil for unit tests.
func NewManager(store storage.WatchStore, ingester Ingester) *Manager {
    return &Manager{
        store:    store,
        ingester: ingester,
        cancels:  make(map[string]context.CancelFunc),
    }
}

// Start loads all active watches from storage and begins watching.
func (m *Manager) Start(ctx context.Context) error {
    watches, err := m.store.ListWatches(ctx, "active")
    if err != nil {
        return err
    }
    for _, cfg := range watches {
        if err := m.startWatch(ctx, cfg); err != nil {
            // Log and continue; don't abort all watches for one bad path.
            _ = err
        }
    }
    <-ctx.Done()
    m.Stop()
    return nil
}

// Stop cancels all active watch goroutines.
func (m *Manager) Stop() {
    m.mu.Lock()
    defer m.mu.Unlock()
    for id, cancel := range m.cancels {
        cancel()
        delete(m.cancels, id)
    }
}

// AddWatch persists a new watch config and starts watching immediately.
func (m *Manager) AddWatch(ctx context.Context, cfg *WatchConfig) error {
    if err := m.store.CreateWatch(ctx, cfg); err != nil {
        return err
    }
    return m.startWatch(ctx, cfg)
}

// RemoveWatch stops watching and deletes the config.
func (m *Manager) RemoveWatch(ctx context.Context, id string) error {
    m.cancelWatch(id)
    return m.store.DeleteWatch(ctx, id)
}

// PauseWatch stops the goroutine and updates status.
func (m *Manager) PauseWatch(ctx context.Context, id string) error {
    m.cancelWatch(id)
    cfg, err := m.store.GetWatch(ctx, id)
    if err != nil {
        return err
    }
    cfg.Status = "paused"
    return m.store.UpdateWatch(ctx, cfg)
}

// ResumeWatch restarts a paused watch.
func (m *Manager) ResumeWatch(ctx context.Context, id string) error {
    cfg, err := m.store.GetWatch(ctx, id)
    if err != nil {
        return err
    }
    cfg.Status = "active"
    if err := m.store.UpdateWatch(ctx, cfg); err != nil {
        return err
    }
    return m.startWatch(ctx, cfg)
}

func (m *Manager) cancelWatch(id string) {
    m.mu.Lock()
    defer m.mu.Unlock()
    if cancel, ok := m.cancels[id]; ok {
        cancel()
        delete(m.cancels, id)
    }
}

func (m *Manager) startWatch(ctx context.Context, cfg *WatchConfig) error {
    watchCtx, cancel := context.WithCancel(ctx)
    m.mu.Lock()
    m.cancels[cfg.ID] = cancel
    m.mu.Unlock()
    go m.watchLoop(watchCtx, cfg)
    return nil
}
```

func (m *Manager) watchLoop(ctx context.Context, cfg *WatchConfig):
    1. Try fsnotify.NewWatcher(); if err != nil OR env CTXT_WATCH_POLLING=1 → go pollLoop(ctx, cfg); return
    2. Walk cfg.Path recursively with filepath.WalkDir; for each dir: watcher.Add(dir)
    3. debounce map[string]*time.Timer (keyed by absolute file path)
    4. Event loop (select on watcher.Events, watcher.Errors, ctx.Done):
       CREATE|WRITE event:
         - Compute relPath = strings.TrimPrefix(absPath, cfg.Path+"/")
         - If !MatchAny(cfg.IncludePatterns, relPath) → skip
         - If MatchAny(cfg.ExcludePatterns, relPath) → skip
         - If event is on a directory: watcher.Add(absPath); continue
         - Debounce: reset timer for absPath; fire after cfg.DebounceMS ms → call m.processFile(ctx, cfg, absPath)
       REMOVE|RENAME event:
         - call m.deleteFile(ctx, cfg, absPath)
       Error: log; continue

func (m *Manager) processFile(ctx context.Context, cfg *WatchConfig, absPath string):
    1. Read file; compute SHA-256 hex hash
    2. GetFileRecord(ctx, cfg.ID, absPath) → existing record (err = not found is OK)
    3. If existing.ContentHash == newHash → skip (no change)
    4. Build AnalyzeRequest: Content=fileContent, Type=detectType(ext), Pipeline=cfg.PipelineOverride, Source="watch:<cfg.Mode>:<relPath>"
    5. jobID, err := m.ingester.Analyze(ctx, req); update record.ObjectID if result available
    6. UpsertFileRecord(ctx, &WatchFileRecord{WatchID: cfg.ID, FilePath: absPath, ContentHash: newHash, LastSeen: now})
    7. If err: store.UpdateWatch with LastError

func (m *Manager) deleteFile(ctx context.Context, cfg *WatchConfig, absPath string):
    1. rec, err := store.GetFileRecord(ctx, cfg.ID, absPath); if not found → return nil
    2. If rec.ObjectID != "": m.ingester.DeleteObject(ctx, rec.ObjectID)
    3. store.DeleteFileRecord(ctx, cfg.ID, absPath)

func detectType(absPath string) string:
    ext := strings.ToLower(filepath.Ext(absPath))
    switch ext {
    case ".md", ".markdown": return "text"
    case ".pdf": return "pdf"
    case ".html", ".htm": return "url"
    default: return "text"
    }

```
func (m *Manager) pollLoop(ctx context.Context, cfg *WatchConfig):
    ticker := time.NewTicker(5 * time.Second)
    defer ticker.Stop()
    seen := map[string]struct{}{}  // tracks files seen in last walk

    for {
        select {
        case <-ctx.Done(): return
        case <-ticker.C:
            current := map[string]struct{}{}
            filepath.WalkDir(cfg.Path, func(path string, d fs.DirEntry, err error) error {
                if d.IsDir() { return nil }
                relPath := strings.TrimPrefix(path, cfg.Path+"/")
                if !MatchAny(cfg.IncludePatterns, relPath) { return nil }
                if MatchAny(cfg.ExcludePatterns, relPath)  { return nil }
                current[path] = struct{}{}
                m.processFile(ctx, cfg, path)  // dedup by hash inside processFile
                return nil
            })
            // Detect deletions: files in seen but not in current.
            for path := range seen {
                if _, ok := current[path]; !ok {
                    m.deleteFile(ctx, cfg, path)
                }
            }
            seen = current
        }
    }

// mode controls how the file is parsed: "obsidian", "logseq", or "generic".
// vaultRoot is the root directory of the watch (used to compute relPath as Source).
func ParseFile(path, mode, vaultRoot string) (service.AnalyzeRequest, error) {
    switch mode {
    case "obsidian":
        return parseObsidianFile(path, vaultRoot)
    case "logseq":
        return parseLogseqFile(path, vaultRoot)
    default:
        return parseGenericFile(path, vaultRoot)

// DetectMode infers the watch mode from the directory structure.
// It checks for ".obsidian/" (Obsidian vault), "logseq/" (Logseq graph),
// and falls back to "generic".
func DetectMode(path string) string {
    if fi, err := os.Stat(filepath.Join(path, ".obsidian")); err == nil && fi.IsDir() {
        return "obsidian"
    }
    if fi, err := os.Stat(filepath.Join(path, "logseq")); err == nil && fi.IsDir() {
        return "logseq"
    }
    return "generic"
}

// CreateWatch handles POST /api/v1/watches.
func CreateWatch(svc *service.Service, mgr *watcher.Manager) http.HandlerFunc {
    return func(w http.ResponseWriter, r *http.Request) {
        var req struct {
            Path             string   `json:"path"`
            Mode             string   `json:"mode"`
            IncludePatterns  []string `json:"include_patterns"`
            ExcludePatterns  []string `json:"exclude_patterns"`
            DebounceMS       int      `json:"debounce_ms"`
            PipelineOverride string   `json:"pipeline_override"`
            Paused           bool     `json:"paused"`
        }
        if err := json.NewDecoder(r.Body).Decode(&req); err != nil {
            WriteError(w, http.StatusBadRequest, "INVALID_REQUEST", "invalid JSON body")
            return
        }
        if req.Path == "" {
            WriteError(w, http.StatusBadRequest, "INVALID_REQUEST", "path is required")
            return
        }
        if req.Mode == "" {
            req.Mode = watcher.DetectMode(req.Path)
        }
        if req.DebounceMS == 0 {
            req.DebounceMS = 500
        }
        status := "active"
        if req.Paused {
            status = "paused"
        }
        now := time.Now()
        cfg := &watcher.WatchConfig{
            ID:               uuid.New().String(),
            Path:             req.Path,
            Mode:             req.Mode,
            IncludePatterns:  req.IncludePatterns,
            ExcludePatterns:  req.ExcludePatterns,
            DebounceMS:       req.DebounceMS,
            Status:           status,
            PipelineOverride: req.PipelineOverride,
            CreatedAt:        now,
            UpdatedAt:        now,
        }
        if mgr != nil {
            if err := mgr.AddWatch(r.Context(), cfg); err != nil {
                WriteError(w, http.StatusInternalServerError, "INTERNAL_ERROR", err.Error())
                return
            }
        } else {
            if err := svc.Store.Watches().CreateWatch(r.Context(), cfg); err != nil {
                WriteError(w, http.StatusInternalServerError, "INTERNAL_ERROR", err.Error())
                return
            }
        }
        WriteJSON(w, http.StatusCreated, cfg)
    }
}

    }
}

//go:build integration

package cmd_test

import (
    "context"
    "os"
    "path/filepath"
    "testing"
    "time"

    "github.com/ideacrafterslabs/ctxt/internal/jobs"
    "github.com/ideacrafterslabs/ctxt/internal/pipeline/builtins"

    "github.com/ideacrafterslabs/ctxt/internal/storageutil"
    "github.com/ideacrafterslabs/ctxt/internal/watcher"
    "net/http/httptest"
)

func TestWatch_E2E(t *testing.T) {
    // 1. Build service + watcher + server.
    driver := storageutil.NewTestDriver(t)
    q := jobs.NewQueue(driver.Jobs())
    pipes := builtins.Registry()
    engine := search.NewEngine(driver)
    svc := service.New(driver, q, pipes, engine, "", nil)
    mgr := watcher.NewManager(driver.Watches(), svc)

    ts := httptest.NewServer(httpserver.NewRouter(svc, mgr))
    defer ts.Close()

    // 2. Create a watch on a temp dir.
    dir := t.TempDir()
    ctx, cancel := context.WithTimeout(context.Background(), 10*time.Second)
    defer cancel()

    cfg := &watcher.WatchConfig{
        ID: "e2e-watch", Path: dir, Mode: "generic",
        IncludePatterns: []string{"**/*.md"},
        ExcludePatterns: []string{},
        DebounceMS: 100, Status: "active",
        CreatedAt: time.Now(), UpdatedAt: time.Now(),
    }
    if err := mgr.AddWatch(ctx, cfg); err != nil {
        t.Fatalf("AddWatch: %v", err)
    }

    // 3. Write a .md file.
    notePath := filepath.Join(dir, "test-note.md")
    os.WriteFile(notePath, []byte("# E2E Test\n\nSome content."), 0644)

    // 4. Wait up to 2s for a file record to appear.
    deadline := time.Now().Add(2 * time.Second)
    var rec *watcher.WatchFileRecord
    for time.Now().Before(deadline) {
        r, err := driver.Watches().GetFileRecord(ctx, "e2e-watch", notePath)
        if err == nil && r.ContentHash != "" {
            rec = r
            break
        }
        time.Sleep(50 * time.Millisecond)
    }
    if rec == nil {
        t.Fatal("file record not created within 2s")
    }
    firstHash := rec.ContentHash

    // 5. Overwrite the file and assert content_hash changed.
    os.WriteFile(notePath, []byte("# E2E Test\n\nUpdated content."), 0644)
    deadline = time.Now().Add(2 * time.Second)
    for time.Now().Before(deadline) {
        r, err := driver.Watches().GetFileRecord(ctx, "e2e-watch", notePath)
        if err == nil && r.ContentHash != firstHash {
            break
        }
        time.Sleep(50 * time.Millisecond)
    }
    updatedRec, err := driver.Watches().GetFileRecord(ctx, "e2e-watch", notePath)
    if err != nil {
        t.Fatalf("get file record after update: %v", err)
    }
    if updatedRec.ContentHash == firstHash {
        t.Error("content_hash should have changed after file update")
    }

    // 6. Delete the file and assert object is removed.
    os.Remove(notePath)
    deadline = time.Now().Add(2 * time.Second)
    for time.Now().Before(deadline) {
        _, err := driver.Watches().GetFileRecord(ctx, "e2e-watch", notePath)
        if err != nil {
            break // record deleted
        }
        time.Sleep(50 * time.Millisecond)
    }
    _, err = driver.Watches().GetFileRecord(ctx, "e2e-watch", notePath)
    if err == nil {
        t.Error("expected file record to be deleted after file removal")
    }
}

## Key design decisions encoded in this plan

1. **No circular import:** `WatchConfig` and `WatchFileRecord` live in `internal/watcher/`, not `internal/storage/`. Storage imports watcher types; watcher imports storage interface. This is the same pattern used for `storage.Feed` (storage defines the type, service imports storage).

2. **`NewRouter` signature change:** Adding `*watcher.Manager` as a second parameter is a one-line diff and keeps handler construction explicit. All existing tests pass `nil` safely because `CreateWatch` and `DeleteWatch` fall back to direct store calls when `mgr == nil`.

3. **Polling is environment-gated:** `CTXT_WATCH_POLLING=1` forces polling (useful for Docker volumes and NFS mounts where inotify is unavailable). The fsnotify init error also triggers the fallback automatically.

4. **Debounce is per-file:** A `map[string]*time.Timer` tracks one timer per absolute path. Rapid successive writes to the same file (e.g. editor auto-save) collapse into a single ingestion call after `cfg.DebounceMS` ms of inactivity.

5. **SHA-256 content dedup:** `processFile` compares the new hash against `WatchFileRecord.ContentHash`. Files that haven't changed (touch with no content diff, symlink refreshes) do not trigger re-ingestion.
