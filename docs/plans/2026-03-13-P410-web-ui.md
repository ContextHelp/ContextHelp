```go
package http

import (
	"net/http"
	"strings"
)

// allowedDevOrigins are extra origins permitted only when devCORS is true.
var allowedDevOrigins = []string{
	"http://localhost:5173",
	"http://127.0.0.1:5173",
}

// corsAllowedPrefixes are always-allowed origin prefixes (extension origins).
var corsAllowedPrefixes = []string{
	"chrome-extension://",
	"moz-extension://",
	"safari-web-extension://",
}

// isLocalhostHost returns true when the request Host header is a localhost address.
func isLocalhostHost(host string) bool {
	// Strip port.
	h := host
	if i := strings.LastIndex(host, ":"); i != -1 {
		h = host[:i]
	}
	return h == "localhost" || h == "127.0.0.1" || h == "::1"
}

// CORS returns middleware that sets Access-Control-* headers for permitted origins.
// When devCORS is true, Vite dev server origins are also allowed.
// CORS is only applied when the request Host is localhost — never for remote hosts.
func CORS(devCORS bool) func(http.Handler) http.Handler {
	return func(next http.Handler) http.Handler {
		return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
			// Only apply CORS on localhost.
			if !isLocalhostHost(r.Host) {
				next.ServeHTTP(w, r)
				return
			}

			origin := r.Header.Get("Origin")
			if origin == "" {
				next.ServeHTTP(w, r)
				return
			}

			allowed := false

			// Always allow extension origins.
			for _, prefix := range corsAllowedPrefixes {
				if strings.HasPrefix(origin, prefix) {
					allowed = true
					break
				}
			}

			// Allow dev origins only when --dev flag is set.
			if !allowed && devCORS {
				for _, o := range allowedDevOrigins {
					if origin == o {
						allowed = true
						break
					}
				}
			}

			if allowed {
				w.Header().Set("Access-Control-Allow-Origin", origin)
				w.Header().Set("Access-Control-Allow-Methods", "GET, POST, PATCH, DELETE, OPTIONS")
				w.Header().Set("Access-Control-Allow-Headers", "Content-Type, Authorization, X-Request-ID")
				w.Header().Set("Access-Control-Expose-Headers", "X-Request-ID")
				w.Header().Set("Vary", "Origin")
			}

			// Handle preflight.
			if r.Method == http.MethodOptions {
				w.WriteHeader(http.StatusNoContent)
				return
			}

			next.ServeHTTP(w, r)
		})
	}
}
```

**Step 1.4 — Create `internal/server/http/handlers_events.go`.**

```go
package http

import (
	"encoding/json"
	"fmt"
	"net/http"

	"github.com/ideacrafterslabs/ctxt/internal/events"
	"github.com/ideacrafterslabs/ctxt/internal/service"
)

// HandleSSE streams all bus events as Server-Sent Events to the client.
// Each event is encoded as a JSON payload in the SSE `data:` field.
// The connection is held open until the client disconnects.
//
// Route: GET /api/v1/events
func HandleSSE(svc *service.Service) http.HandlerFunc {
	return func(w http.ResponseWriter, r *http.Request) {
		// Verify the client supports streaming.
		flusher, ok := w.(http.Flusher)
		if !ok {
			WriteError(w, http.StatusInternalServerError, "STREAMING_UNSUPPORTED", "streaming not supported")
			return
		}

		// Set SSE headers.
		w.Header().Set("Content-Type", "text/event-stream")
		w.Header().Set("Cache-Control", "no-cache")
		w.Header().Set("Connection", "keep-alive")
		w.Header().Set("X-Accel-Buffering", "no")
		w.WriteHeader(http.StatusOK)

		// Channel for events delivered to this connection.
		ch := make(chan events.Event, 64)

		// Subscribe to all events via wildcard.
		svc.Bus.Subscribe("*", func(_ interface{}, e events.Event) error {
			select {
			case ch <- e:
			default:
				// Drop on full buffer rather than blocking the bus.
			}
			return nil
		})

		// Stream events until client disconnects.
		ctx := r.Context()
		for {
			select {
			case <-ctx.Done():
				return
			case e := <-ch:
				data, err := json.Marshal(e)
				if err != nil {
					continue
				}
				fmt.Fprintf(w, "data: %s\n\n", data)
				flusher.Flush()
			}
		}
	}
}
```

> Note: `events.Handler` signature is `func(ctx context.Context, e Event) error`. The handler above uses a channel-bridge pattern to decouple bus delivery from HTTP writes. Adjust the `Subscribe` call signature to match `func(ctx context.Context, e events.Event) error` exactly:

```go
svc.Bus.Subscribe("*", func(ctx context.Context, e events.Event) error {
    select {
    case ch <- e:
    default:
    }
    return nil
})
```

```go
	bus := events.NewLocalBus()
	svc := &service.Service{Bus: bus}

	handler := httpserver.HandleSSE(svc)

	req := httptest.NewRequest(http.MethodGet, "/api/v1/events", nil)
	ctx, cancel := context.WithTimeout(req.Context(), 2*time.Second)
	defer cancel()
	req = req.WithContext(ctx)

	// Use a pipe-backed recorder that supports Flush.
	pr, pw := io.Pipe()
	rr := &flushRecorder{pw: pw}

	done := make(chan struct{})
	go func() {
		defer close(done)
		handler.ServeHTTP(rr, req)
	}()

	// Give SSE handler time to subscribe.
	time.Sleep(20 * time.Millisecond)

	// Publish a test event.
	ev, _ := events.NewEvent("test.source", "job.completed", map[string]string{"id": "job_001"})
	bus.Publish(context.Background(), ev)

	// Read one SSE line from the pipe.
	scanner := bufio.NewScanner(pr)
	var line string
	for scanner.Scan() {
		line = scanner.Text()
		if strings.HasPrefix(line, "data:") {
			break
		}
	}

	cancel() // disconnect client
	<-done
```

```typescript
export class ApiError extends Error {
  constructor(
    public readonly status: number,
    public readonly code: string,
    message: string
  ) {
    super(message);
    this.name = "ApiError";
  }
}

export async function apiFetch<T>(
  path: string,
  options?: RequestInit
): Promise<T> {
  const res = await fetch(`${BASE}${path}`, {
    ...options,
    headers: {
      "Content-Type": "application/json",
      ...options?.headers,
    },
  });

  if (!res.ok) {
    let code = "HTTP_ERROR";
    let message = `HTTP ${res.status}`;
    try {
      const body = (await res.json()) as { code?: string; message?: string };
      code = body.code ?? code;
      message = body.message ?? message;
    } catch {
      // ignore parse errors
    }
    throw new ApiError(res.status, code, message);
  }

  // 204 No Content
  if (res.status === 204) {
    return undefined as unknown as T;
  }

  return res.json() as Promise<T>;
}
```

**Step 6.4 — Create `web/ui/src/hooks/useJobEvents.ts`.**

```typescript
import { useEffect, useRef } from "react";
import { useQueryClient } from "@tanstack/react-query";

type SSEEventType = "job.completed" | "job.failed" | "job.enqueued" | "object.updated" | "object.deleted";

/**
 * Subscribes to the SSE stream at /api/v1/events.
 * Invalidates relevant TanStack Query cache keys on job lifecycle events.
 * Mount once in AppShell — it is headless (renders nothing).
 */
export function useJobEvents() {
  const queryClient = useQueryClient();
  const esRef = useRef<EventSource | null>(null);

  useEffect(() => {
    const es = new EventSource("/api/v1/events");
    esRef.current = es;

    es.onmessage = (e: MessageEvent<string>) => {
      try {
        const event = JSON.parse(e.data) as { type: SSEEventType; data?: { id?: string } };
        switch (event.type) {
          case "job.enqueued":
          case "job.completed":
          case "job.failed":
            queryClient.invalidateQueries({ queryKey: ["jobs"] });
            break;
          case "object.updated":
          case "object.deleted":
            queryClient.invalidateQueries({ queryKey: ["objects"] });
            queryClient.invalidateQueries({ queryKey: ["search"] });
            break;
        }
      } catch {
        // ignore malformed events
      }
    };

    es.onerror = () => {
      // EventSource auto-reconnects. Log in dev only.
      if (import.meta.env.DEV) {
        console.warn("[SSE] connection error — browser will retry");
      }
    };

    return () => {
      es.close();
      esRef.current = null;
    };
  }, [queryClient]);
}
```

```tsx
import { useEffect, useRef, useState } from "react";
import { useCaptureModal } from "@/store/ui";
import { useCapture } from "@/hooks/useCapture";
import toast from "react-hot-toast";

const PIPELINES = [
  "text.default",
  "url.article",
  "url.generic",
  "text.long",
  "code.snippet",
];

export function CaptureModal() {
  const { isOpen, close } = useCaptureModal();
  const [content, setContent] = useState("");
  const [pipeline, setPipeline] = useState("");
  const [hints, setHints] = useState("");
  const textareaRef = useRef<HTMLTextAreaElement>(null);
  const { mutate: capture, isPending } = useCapture();

  // Global Cmd+K / Ctrl+K shortcut.
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if ((e.metaKey || e.ctrlKey) && e.key === "k") {
        e.preventDefault();
        useCaptureModal.getState().open();
      }
    };
    document.addEventListener("keydown", handler);
    return () => document.removeEventListener("keydown", handler);
  }, []);

  // Focus textarea when modal opens.
  useEffect(() => {
    if (isOpen) {
      setTimeout(() => textareaRef.current?.focus(), 50);
    } else {
      setContent("");
      setPipeline("");
      setHints("");
    }
  }, [isOpen]);

  // Escape to close.
  useEffect(() => {
    const handler = (e: KeyboardEvent) => {
      if (e.key === "Escape") close();
    };
    document.addEventListener("keydown", handler);
    return () => document.removeEventListener("keydown", handler);
  }, [close]);

  if (!isOpen) return null;

  const handleSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!content.trim()) return;

    const isURL = content.trim().startsWith("http");
    capture(
      {
        content: content.trim(),
        type: isURL ? "url" : "text",
        pipeline: pipeline || undefined,
        source: hints || undefined,
      },
      {
        onSuccess: (res) => {
          toast.success(`Queued job ${res.job_id}`);
          close();
        },
        onError: () => {
          toast.error("Capture failed. Try again.");
        },
      }
    );
  };

  return (
    <>
      {/* Backdrop */}
      <div
        aria-hidden="true"
        onClick={close}
        style={{
          position: "fixed",
          inset: 0,
          background: "rgba(26,35,50,0.6)",
          zIndex: 200,

          borderRadius: "var(--radius-lg)",
          boxShadow: "var(--shadow-modal)",
          zIndex: 201,
          padding: "var(--spacing-lg)",
          display: "flex",
          flexDirection: "column",
          gap: "var(--spacing-md)",
        }}
      >
        <h2 id="capture-modal-title" style={{ margin: 0, fontSize: "1.1rem" }}>
          Capture
        </h2>

        <form onSubmit={handleSubmit} style={{ display: "flex", flexDirection: "column", gap: "var(--spacing-sm)" }}>
          <label htmlFor="capture-content" style={{ fontSize: 13, fontWeight: 600 }}>
            Content or URL
          </label>
          <textarea
            id="capture-content"
            ref={textareaRef}
            value={content}
            onChange={(e) => setContent(e.target.value)}
            placeholder="Paste a URL, text, code snippet, or anything…"
            rows={5}
            required
            aria-required="true"
            style={{
              width: "100%",
              padding: 10,
              borderRadius: "var(--radius-md)",
              border: "1px solid var(--color-slate)",
              fontFamily: "var(--font-sans)",
              fontSize: 14,
              resize: "vertical",
              background: "transparent",
              color: "var(--color-ink)",

          <label htmlFor="capture-pipeline" style={{ fontSize: 13, fontWeight: 600 }}>

          </label>

            value={pipeline}
            onChange={(e) => setPipeline(e.target.value)}
            style={{ padding: "6px 10px", borderRadius: "var(--radius-md)", border: "1px solid var(--color-slate)", background: "transparent", color: "var(--color-ink)", fontSize: 14 }}
          >
            <option value="">Auto-detect</option>
```

```tsx
export function GraphCanvas({ entities }: GraphCanvasProps) {
  const nodes: Node[] = useMemo(
    () =>
      entities.map((e, i) => ({
        id: e.slug,
        type: "entity",
        position: {
          x: (i % 8) * 160 + 40,
          y: Math.floor(i / 8) * 120 + 40,
        },
        data: { label: e.title ?? e.slug, namespace: e.namespace },
      })),

            style: { stroke: theme.colors.accent, strokeWidth: 1 },
          };
        })
        .filter(Boolean) as Edge[],
    [entities]
  );

  return (
    <ReactFlow
      nodes={nodes}
      edges={edges}
      nodeTypes={nodeTypes}
      fitView
      aria-label="Knowledge graph visualization"
```

### Task 13: Compose + Settings

**Goal:** Template-based composition interface and a settings page for preferences.

**Step 13.1 — Create `web/ui/src/components/compose/ComposeView.tsx`.**

```tsx
import { useState } from "react";
import { useObjects } from "@/hooks/useObjects";
import { apiFetch } from "@/api/client";
import toast from "react-hot-toast";

const COMPOSITION_TYPES = ["brief", "plan", "summary", "draft"];

export function ComposeView() {
  const [compType, setCompType] = useState("brief");
  const [selectedIDs, setSelectedIDs] = useState<string[]>([]);
  const [output, setOutput] = useState<string>("");
  const [isLoading, setIsLoading] = useState(false);

  const { data } = useObjects({ limit: 50 });
  const objects = data?.items ?? [];

  const toggle = (id: string) => {
    setSelectedIDs((prev) =>
      prev.includes(id) ? prev.filter((x) => x !== id) : [...prev, id]
    );
  };

  const generate = async () => {
    if (!selectedIDs.length) {
      toast.error("Select at least one knowledge object.");
      return;
    }
    setIsLoading(true);
    try {
      const res = await apiFetch<{ content: string }>("/api/v1/compose", {
        method: "POST",
        body: JSON.stringify({ object_ids: selectedIDs, type: compType }),
      });
      setOutput(res.content);
    } catch {
      toast.error("Compose failed.");
    } finally {
      setIsLoading(false);
    }
  };

  return (
    <div style={{ display: "flex", gap: "var(--spacing-lg)", height: "calc(100vh - 120px)" }}>
      {/* Left panel: selection */}
      <div style={{ width: 320, display: "flex", flexDirection: "column", gap: "var(--spacing-md)" }}>
        <h1 style={{ fontSize: "1.4rem", margin: 0 }}>Compose</h1>

        <label htmlFor="comp-type" style={{ fontSize: 13, fontWeight: 600 }}>Type</label>
        <select
          id="comp-type"
          value={compType}
          onChange={(e) => setCompType(e.target.value)}
          style={{ padding: "6px 10px", borderRadius: "var(--radius-md)", border: "1px solid var(--color-slate)", fontSize: 14 }}
        >
          {COMPOSITION_TYPES.map((t) => (
            <option key={t} value={t}>{t}</option>
          ))}
        </select>

        <fieldset style={{ border: "1px solid var(--color-slate)", borderRadius: "var(--radius-md)", padding: "var(--spacing-sm)" }}>
          <legend style={{ fontSize: 13, fontWeight: 600, padding: "0 4px" }}>
            Objects ({selectedIDs.length} selected)
          </legend>
          <ul style={{ listStyle: "none", padding: 0, margin: 0, maxHeight: 400, overflowY: "auto" }}>
            {objects.map((obj) => {
              const title =
                (obj.metadata?.title as string | undefined) ??
                obj.summaries?.[0]?.slice(0, 60) ??
                obj.id;
              return (
                <li key={obj.id}>
                  <label style={{ display: "flex", alignItems: "center", gap: 8, padding: "4px 0", fontSize: 13, cursor: "pointer" }}>
                    <input
                      type="checkbox"
                      checked={selectedIDs.includes(obj.id)}
                      onChange={() => toggle(obj.id)}
                      aria-label={`Select: ${title}`}
                    />
                    <span style={{ overflow: "hidden", textOverflow: "ellipsis", whiteSpace: "nowrap" }}>{title}</span>
                  </label>
                </li>
              );
            })}
          </ul>
        </fieldset>

        <button
          onClick={generate}
          disabled={isLoading}
          style={{
            padding: "10px 20px",
            borderRadius: "var(--radius-md)",
            border: "none",
            background: "var(--color-accent)",
            color: "var(--color-ink)",
            fontWeight: 600,
            fontSize: 14,
            cursor: isLoading ? "wait" : "pointer",
          }}
        >
          {isLoading ? "Generating…" : "Generate"}
        </button>
      </div>

      {/* Right panel: output */}
      <div style={{ flex: 1, display: "flex", flexDirection: "column", gap: "var(--spacing-sm)" }}>
        <h2 style={{ fontSize: "1rem", margin: 0, color: "var(--color-slate)" }}>Output</h2>
        {output ? (
          <pre style={{
            flex: 1,
            padding: "var(--spacing-md)",
            borderRadius: "var(--radius-lg)",
            border: "1px solid var(--color-slate)",
            fontFamily: "var(--font-mono)",
            fontSize: 13,
            whiteSpace: "pre-wrap",
            overflowY: "auto",
            background: "#f6f8fa",
          }}>
            {output}
          </pre>
        ) : (
          <div style={{ flex: 1, display: "flex", alignItems: "center", justifyContent: "center", border: "1px dashed var(--color-slate)", borderRadius: "var(--radius-lg)" }}>
            <p style={{ color: "var(--color-slate)", fontStyle: "italic" }}>
              Select objects and click Generate.
            </p>
          </div>
        )}
        {output && (
          <button
            onClick={() => {
              navigator.clipboard.writeText(output);
              toast.success("Copied to clipboard");
            }}
            style={{ alignSelf: "flex-end", padding: "6px 14px", borderRadius: "var(--radius-md)", border: "1px solid var(--color-slate)", background: "transparent", cursor: "pointer", fontSize: 13 }}
          >
            Copy
          </button>
        )}
      </div>
    </div>
  );
}
```
